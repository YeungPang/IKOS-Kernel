workflow_case_is_overdue(_status, _now_ts, _sla_due_ts) ⦃
    ↲((_status ≠ closed) ⋀ (_status ≠ cancelled) ⋀ (_now_ts > _sla_due_ts))
⦄

approval_is_final(_decision, _escalation_level) ⦃
    ↲(((_decision = approved) ⋁ (_decision = rejected)) ⋀ (_escalation_level = 0))
⦄

workflow_case_is_overdue_by_ref(_case_ref, _now_ts) ⦃
    (_status ≔ workflow_case_status_by_ref[_case_ref]) ⋀
    (_sla_due_ts ≔ workflow_case_sla_due_by_ref[_case_ref]) ⋀

approval_is_final_by_ref(_approval_ref) ⦃
    (_decision ≔ approval_decision_by_ref[_approval_ref]) ⋀
    (_escalation_level ≔ approval_escalation_level_by_ref[_approval_ref]) ⋀
