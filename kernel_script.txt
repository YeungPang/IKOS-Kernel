workflow_transition_policy(_payload) ⦃
    (_entity_type ≔ _payload[entity_type]) ⋀
    (_to_state ≔ _payload[to_state]) ⋀
    (_event ≔ _payload[event]) ⋀
    (_actor_ref ≔ _payload[actor_ref]) ⋀
    (_entity_type = "workflow_case") ⋀
    ((_to_state = "approved") ⋁ (_to_state = "rejected")) ⋀
    (_event ≠ "approval_recorded") ⋀
    ↲({
        mode: deny,
        reason: workflow_case_terminal_state_requires_approval_event,
        allowed: false,
        message: "workflow_case can move to approved/rejected only from approval_recorded event"
    })
⦄

workflow_transition_policy(_payload) ⦃
    (_entity_type ≔ _payload[entity_type]) ⋀
    (_to_state ≔ _payload[to_state]) ⋀
    (_event ≔ _payload[event]) ⋀
    (_actor_ref ≔ _payload[actor_ref]) ⋀
    (_entity_type = "workflow_case") ⋀
    ((_to_state = "approved") ⋁ (_to_state = "rejected")) ⋀
    (_actor_ref = "system") ⋀
    ↲({
        mode: deny,
        reason: manual_terminal_transition_not_allowed,
        allowed: false,
        message: "manual transition to terminal workflow_case state is not allowed"
    })
⦄

# SQL plan templates emitted by SOLF, executed by Python adapters.
sql_plan_create_workflow_case(_args) ⦃
    (_case_no ≔ _args[case_no]) ⋀
    (_case_type ≔ _args[case_type]) ⋀
    (_subject_ref ≔ _args[subject_ref]) ⋀
    (_initiated_by_ref ≔ _args[initiated_by_ref]) ⋀
    (_current_step ≔ _args[current_step]) ⋀
    (_sla_due_at ≔ _args[sla_due_at]) ⋀
    (_status ≔ _args[status]) ⋀
    ↲({
        operation: insert,
        table: workflow_cases,
        values: {
            case_no: _case_no,
            case_type: _case_type,
            subject_ref: _subject_ref,
            initiated_by_ref: _initiated_by_ref,
            current_step: _current_step,
            sla_due_at: _sla_due_at,
            status: _status
        },
        returning: [id]
    })
⦄
